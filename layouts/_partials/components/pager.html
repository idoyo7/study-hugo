{{/* Article navigation scoped to the current page's direct siblings. */}} {{- $reversePagination := .Store.Get "reversePagination" | default false -}} {{- $prev := false -}} {{- $next := false -}} {{- $prevExplicit := false -}} {{- $nextExplicit := false -}} {{- /* PrevInSection/NextInSection flatten every page below a top-level section. With the four-domain information architecture that would connect unrelated groups. Walk only the current parent's visible direct children instead. */ -}} {{- with .Parent -}} {{- $siblings := where .Pages.ByWeight "Params.sidebar.exclude" "!=" true -}} {{- $currentIndex := -1 -}} {{- range $index, $sibling := $siblings -}} {{- if eq $sibling.RelPermalink $.RelPermalink -}} {{- $currentIndex = $index -}} {{- end -}} {{- end -}} {{- if ge $currentIndex 0 -}} {{- if gt $currentIndex 0 -}} {{- $prev = index $siblings (sub $currentIndex 1) -}} {{- end -}} {{- if lt (add $currentIndex 1) (len $siblings) -}} {{- $next = index $siblings (add $currentIndex 1) -}} {{- end -}} {{- end -}} {{- end -}} {{- if $reversePagination -}} {{- $orderedPrev := $prev -}} {{- $prev = $next -}} {{- $next = $orderedPrev -}} {{- end -}} {{- /* Preserve Hextra's explicit page-level pager overrides. */ -}} {{- if eq .Params.prev false }} {{- if $reversePagination }}{{ $next = false }}{{ else }}{{ $prev = false }}{{ end -}} {{ else }} {{- with .Params.prev -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $next = . }}{{ $nextExplicit = true }}{{ else }}{{ $prev = . }}{{ $prevExplicit = true }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq .Params.next false }} {{- if $reversePagination }}{{ $prev = false }}{{ else }}{{ $next = false }}{{ end -}} {{ else }} {{- with .Params.next -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $prev = . }}{{ $prevExplicit = true }}{{ else }}{{ $next = . }}{{ $nextExplicit = true }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
