{{- /* vm-flow: 애니메이션 플로우 다이어그램. 사용 (권장): {{< flow src="_flow/이름.json" />}} — 스펙은 page bundle 리소스로 분리 사용 (인라인): {{< flow caption="..." >}}{ "nodes":[...], "edges":[...] }{{< /flow >}} src를 쓰면 caption은 JSON 최상위 "caption" 키에서 읽는다. caption 파라미터를 주면 그쪽이 이긴다. 엔진/스타일은 /flow/flow.js, /flow/flow.css (custom/head-end.html에서 로드) */ -}} {{- $caption := .Get "caption" -}} {{- $spec := "" -}} {{- with .Get "src" -}} {{- $res := $.Page.Resources.GetMatch . -}} {{- if not $res -}} {{- errorf "flow: %q 리소스를 찾을 수 없다 — %s" . $.Page.Path -}} {{- end -}} {{- $spec = $res.Content -}} {{- if not $caption -}} {{- $caption = index ($spec | transform.Unmarshal) "caption" -}} {{- end -}} {{- else -}} {{- $spec = $.Inner -}} {{- end -}} {{- $data := $spec | transform.Unmarshal -}} {{- /* groups·edges 는 label 이 없는 항목이 섞여 있고 아예 하나도 없는 스펙도 있다 (edges 배열은 106개 스펙 전부에 있지만 그중 20개는 라벨을 가진 간선이 없다). 리스트를 먼저 열고 안에서 걸러내면 그 20개가 빈
    로 남으므로, 낼 것이 있는지부터 확인하고 리스트를 연다. nodes 는 label 이 필수라 그대로 둔다. */ -}} {{- $gl := slice }} {{- range $data.groups }}{{ with .label }}{{ $gl = $gl | append . }}{{ end }}{{ end }} {{- $el := slice }} {{- range $data.edges }}{{ with .label }}{{ $el = $el | append . }}{{ end }}{{ end -}}
