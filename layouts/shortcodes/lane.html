{{- /* vm-lane: 가로축(시점/연속) + 레인별 막대 다이어그램. 사용 (권장): {{< lane src="_lane/이름.json" />}} — 스펙은 page bundle 리소스로 분리 사용 (인라인): {{< lane caption="..." >}}{ "axis":{...}, "lanes":[...] }{{< /lane >}} src를 쓰면 caption은 JSON 최상위 "caption" 키에서 읽는다. caption 파라미터를 주면 그쪽이 이긴다. 엔진/스타일은 /flow/lane.js, /flow/lane.css (custom/head-end.html에서 로드) */ -}} {{- $caption := .Get "caption" -}} {{- $spec := "" -}} {{- with .Get "src" -}} {{- $res := $.Page.Resources.GetMatch . -}} {{- if not $res -}} {{- errorf "lane: %q 리소스를 찾을 수 없다 — %s" . $.Page.Path -}} {{- end -}} {{- $spec = $res.Content -}} {{- if not $caption -}} {{- $caption = index ($spec | transform.Unmarshal) "caption" -}} {{- end -}} {{- else -}} {{- $spec = $.Inner -}} {{- end -}}
